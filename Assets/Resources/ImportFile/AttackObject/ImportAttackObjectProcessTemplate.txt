using UnityEngine;

public class #CLASS_NAME#Process
{
    private GameObject #CLASS_VARIABLE_NAME#Object;
    private Transform followObjectTransform;
    private Animator animator;
    private AnimatorStateInfo animatorStateInfo; //アニメーションの状態
    private Quaternion followObjectRotation;
    private bool isStartAnime = default;
    private bool isCollision = default;
    private string trantitionTagName;
    private string attackAnimationName;
    private float currentPosiotionXValue = default;
    private float followObjectDistance = 1f; //追従対象との距離
    private float curentXPosition = default;
    private float attackPower = default;
    private bool isCordinateLock = default;


    public #CLASS_NAME#Process(Animator animator, GameObject #CLASS_VARIABLE_NAME#Object)
    {
        this.animator = animator;
        this.#CLASS_VARIABLE_NAME#Object = #CLASS_VARIABLE_NAME#Object;

        SetTagName(AttackEffectAnimationTriggerName.AttackBool.ToString(), AttackEffectAnimatorName.Attack.ToString());
    }

    private void TargetObjectFollow()
    {
        //向いている方向に飛ばす
        if (currentPosiotionXValue < followObjectTransform.position.x)
        {
            curentXPosition = followObjectTransform.position.x + followObjectDistance;

            #CLASS_VARIABLE_NAME#Object.transform.rotation = followObjectRotation = Quaternion.Euler(0, 0, 0);
        }
        else if (currentPosiotionXValue > followObjectTransform.position.x)
        {
            curentXPosition = followObjectTransform.position.x - followObjectDistance;

            #CLASS_VARIABLE_NAME#Object.transform.rotation = followObjectRotation = Quaternion.Euler(0, 180, 0);
        }
        else
        {
            curentXPosition = (followObjectRotation.y > 0) ? followObjectTransform.position.x - followObjectDistance : followObjectTransform.position.x + followObjectDistance;
            #CLASS_VARIABLE_NAME#Object.transform.rotation = followObjectRotation;
        }
        //対象を追従
        #CLASS_VARIABLE_NAME#Object.gameObject.transform.position = new Vector3(
            curentXPosition,
            followObjectTransform.position.y,
            followObjectTransform.position.z);

        //最新のX座標を取得
        //追従しているオブジェクトのX座標を取得
        currentPosiotionXValue = followObjectTransform.transform.position.x;
    }

    public void Attack()
    {
        if (!isCordinateLock)
        {
            //座標を確定する
            TargetObjectFollow();
            isCordinateLock = true;
        }
        if (!isStartAnime)
        {
            //アニメーションを開始
            animator.SetBool(trantitionTagName, true);
            isStartAnime = true;
        }
        //現在のアニメーションのステートを取得ー＞想定：Attack
        animatorStateInfo = animator.GetCurrentAnimatorStateInfo(0);
        //再生されているアニメーションが指定のものかつ、90%終わっているなら終了処理を行う
        if (animatorStateInfo.IsName(attackAnimationName) && animatorStateInfo.normalizedTime > 0.9f) // && animatorStateInfo.normalizedTime > 0.9f
        {
            //アニメーションを終了する
            animator.SetBool(trantitionTagName, false);
            
            isStartAnime = false;

            //現在のアニメーションのステートを取得ー＞想定：Attack_Stand
            animatorStateInfo = animator.GetCurrentAnimatorStateInfo(0);
        }
    }

    public void SetParameter(float power)
    {
        this.attackPower = power;
    }

    public void SetPlayerObjectReference(GameObject playerObject)
    {
        followObjectTransform = playerObject.transform;
    }

    private void SetTagName(string trantitionTagName, string attackAnimationName)
    {
        this.trantitionTagName = trantitionTagName;
        this.attackAnimationName = attackAnimationName;
    }

    public void CollisionEnemy(Collider2D collision)
    {
        if (collision.TryGetComponent<IEnemy>(out var damageable))
        {
            damageable.TakeDamage(attackPower);
        }
    }
}
